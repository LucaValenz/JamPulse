@startuml Sequence_CreatePost
autonumber
skinparam BoxPadding 10
skinparam ParticipantPadding 10

actor "Utente" as User
participant "Frontend (React)" as FE
participant "Middleware JWT" as Auth
participant "Backend (Express)" as BE
database "MongoDB" as DB

User -> FE: Compila il post e pubblica
FE -> BE: POST /api/v1/posts + Header: Bearer <accessToken>
activate BE

BE -> Auth: verifyToken(req, res, next)
activate Auth
Auth -> Auth: Verifica validità token JWT (HS256) e ricava req.user (id, role)
Auth --> BE: Token valido (prosegue)
deactivate Auth

BE -> BE: Validazione input (controllo testo obbligatorio)

alt Dati validi
    BE -> DB: Post.create({ userID: req.user.id, content, media })
    activate DB
    DB --> BE: Conferma salvataggio del post
    deactivate DB
    BE --> FE: Restituisce 201 Created + dati del post
    FE --> User: Aggiorna il Feed mostrando il nuovo post
else Dati non validi
    BE --> FE: Restituisce 400 Bad Request
    FE --> User: Mostra messaggio di errore
end

deactivate BE
@enduml
