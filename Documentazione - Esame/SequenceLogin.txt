@startuml Sequence_Login
autonumber
skinparam BoxPadding 10
<style>
participant {
    Padding 10
}
</style>

actor "Utente" as User
participant "Frontend (React)" as FE
participant "Backend (Express)" as BE
database "MongoDB" as DB

User -> FE: Inserisce username e password e clicca Login
FE -> BE: POST /api/v1/auth/login { username, password }
activate BE

BE -> BE: Valida presenza di username e password
alt Dati mancanti o username vuoto
    BE --> FE: 400 Bad Request
    FE --> User: Mostra messaggio di errore
else Dati presenti
    BE -> DB: User.findOne({ username: username.trim() })
    DB --> BE: Documento utente (hash password e role) oppure null
    opt Utente trovato
        BE -> BE: user.comparePassword(password) (bcrypt)
    end
    alt Utente trovato e password corretta
        BE -> BE: Genera Access Token (durata default 15m, HS256, payload: userId, role)
        BE -> BE: Genera Refresh Token firmato HS256 con userId e jti univoco\nDurata default 7 giorni; ricava expiresAt da exp
        BE -> DB: new RefreshToken({ token, user, expiresAt }).save()
        activate DB
        DB --> BE: Salva Refresh Token (indice TTL su expiresAt)
        deactivate DB
        BE --> FE: 200 OK + { token, accessToken, refreshToken, user }\ntoken è un alias di accessToken
        FE -> FE: Salva nel localStorage: token, refreshToken, user\nAggiorna user e token in AuthContext
        FE --> User: Reindirizzamento alla Home
    else Utente assente o password errata
        BE --> FE: Restituisce 401 Unauthorized (messaggio generico di sicurezza)
        FE --> User: Mostra messaggio di errore
    end
end

deactivate BE
@enduml
