@startuml Sequence_Login
autonumber
skinparam BoxPadding 10
skinparam ParticipantPadding 10

actor "Utente" as User
participant "Frontend (React)" as FE
participant "Backend (Express)" as BE
database "MongoDB" as DB

User -> FE: Inserisce username e password e clicca Login
FE -> BE: POST /api/v1/auth/login { username, password }
activate BE

BE -> DB: User.findOne({ username })
activate DB
DB --> BE: Restituisce documento utente (con hash password e role)
deactivate DB

BE -> BE: user.comparePassword(password) (bcrypt)

alt Credenziali valide
    BE -> BE: Genera Access Token (JWT 15m, HS256, payload: userId, role)
    BE -> DB: RefreshToken.create({ token, user, expiresAt })
    activate DB
    DB --> BE: Salva Refresh Token (con indice TTL 7 giorni)
    deactivate DB
    BE --> FE: Restituisce 200 OK + { accessToken, refreshToken, user }
    FE -> FE: Salva accessToken, refreshToken e dati utente (localStorage)
    FE --> User: Reindirizzamento alla Home
else Credenziali errate
    BE --> FE: Restituisce 401 Unauthorized (messaggio generico di sicurezza)
    FE --> User: Mostra messaggio di errore
end

deactivate BE
@enduml
